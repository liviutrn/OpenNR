cmake_minimum_required(VERSION 3.20)
project(DuoViewIndependentTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(DuoViewGeometryTests stage2_geometry_tests.cpp)
add_executable(DuoViewProbe DuoViewProbe.cpp)
add_executable(DuoViewGazeTests stage4_gaze_tests.cpp)
add_executable(DuoViewFrameTests stage5_frame_tests.cpp)
if(WIN32)
  add_executable(DuoViewWarpTests DuoViewWarpTest.cpp)
  target_link_libraries(DuoViewWarpTests PRIVATE d3d12 dxgi d3dcompiler)
endif()
enable_testing()
add_test(NAME geometry COMMAND DuoViewGeometryTests)
add_test(NAME gaze COMMAND DuoViewGazeTests)
add_test(NAME frame COMMAND DuoViewFrameTests)
